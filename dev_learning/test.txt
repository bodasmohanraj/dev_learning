-- 4) Useful practice queries
SELECT * FROM parks_and_recreation.customers;
SELECT * FROM parks_and_recreation.products;
SELECT * FROM parks_and_recreation.orders;

-- Example 1: Join customers with orders
SELECT c.customer_id, c.first_name, c.country, o.order_id, o.total_amount
FROM parks_and_recreation.customers c
JOIN parks_and_recreation.orders o ON c.customer_id = o.customer_id;

-- Example 2: Filter active customers and high spenders
SELECT *
FROM parks_and_recreation.customers
WHERE is_active = TRUE AND annual_spend > 10000
ORDER BY annual_spend DESC;

-- Example 3: Products with good rating and in stock
SELECT product_name, category, price, rating
FROM parks_and_recreation.products
WHERE is_in_stock = TRUE AND rating >= 4.5
ORDER BY price DESC;

-- Example 4: Count customers by country
SELECT country, COUNT(*) AS total_customers
FROM parks_and_recreation.customers
GROUP BY country
ORDER BY total_customers DESC;
